\section{Chapter 1: Introduction}

\begin{frame}
    \frametitle{Research Title \& Background}
    \begin{block}{Research Title}
        \footnotesize\textbf{Modelling Physical Climate Risk in IFRS 9 ECL Frameworks:\\
        A Two-Regime Markov-Switching Approach to Credit Migration in Zimbabwe's Microfinance Sector}
    \end{block}
    \vspace{0.3em}
    \begin{itemize}
        \item \textbf{IFRS 9 (2018):} Replaced backward-looking incurred-loss accounting with a \textbf{forward-looking ECL} framework (IASB, 2014).
        \item EBA (2023) and NGFS require institutions to integrate \textbf{physical climate risk} into ECL outputs.
        \item \textbf{Zimbabwe's vulnerability:}
        \begin{itemize}
            \item 42\% of MFI client base are smallholder farmers --- directly exposed to drought.
            \item RBZ Portfolio-at-Risk: \textbf{18.7\%} --- more than triple the prudential threshold (RBZ, 2024).
            \item Drought causes highly \textit{covariant} defaults across the entire portfolio.
        \end{itemize}
        \item \textbf{Gap:} No exogenous, climate-driven Regime-Switching Model exists for Zimbabwean MFI credit migration.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Statement of the Problem}
    \begin{block}{Core Conflict}
        The \textbf{dynamic, non-linear reality} of physical climate risk (drought) versus the \textbf{static assessment methodologies} used by Zimbabwean MFIs for credit risk management.
    \end{block}
    \vspace{0.4em}
    \begin{itemize}
        \item Static (TTC) models assume \textit{average} default probabilities --- they cannot capture the abrupt jump in defaults when the system switches into a Drought-Stress regime.
        \item This structural failure produces two critical risks:
    \end{itemize}
    \begin{enumerate}
        \item \textbf{Chronic underestimation} of actual credit loss exposure during climate stress.
        \item \textbf{Inadequate capital provisioning}, threatening institutional solvency and financial inclusion.
    \end{enumerate}
    \vspace{0.4em}
    The solution requires an \textbf{Exogenous Two-Regime Markov-Switching Model} that translates a measurable climate index (SPEI) into a quantifiable, probabilistic switch in the MFI's credit migration process.
\end{frame}

\begin{frame}
    \frametitle{Research Objectives}
    \begin{enumerate}
        \item \textbf{Regime Classification:} Use the SPEI-3 drought index to classify the historical period (2014--2024) into Normal and Drought-Stress credit regimes.
        \item \textbf{Matrix Estimation:} Estimate regime-conditional credit migration matrices ($\hat{M}_{\text{normal}}$ and $\hat{M}_{\text{stress}}$) via Maximum Likelihood Estimation.
        \item \textbf{Model Validation:} Perform a Likelihood Ratio Test (LRT) and out-of-sample discrimination analysis to validate the two-regime structure over a static baseline.
        \item \textbf{Risk Forecasting:} Apply the model to produce a climate-adjusted ECL under multiple IFRS 9-compliant scenarios.
    \end{enumerate}
\end{frame}

\begin{frame}
    \frametitle{Research Questions}
    \begin{enumerate}
        \item How can the SPEI drought index be used to statistically classify the historical time series into distinct Normal and Drought-Stress credit regimes for Zimbabwean MFIs?
        \item What are the quantitative differences between $\hat{M}_{\text{normal}}$ and $\hat{M}_{\text{stress}}$, and how significantly does the Probability of Default differ between regimes?
        \item How can the Exogenous Two-Regime Markov-Switching Model be calibrated via MLE to link climate dynamics to credit migration?
        \item To what extent does the regime-switching model improve ECL estimation relative to a time-homogeneous model for IFRS 9 compliance over a five-year horizon?
    \end{enumerate}
\end{frame}

\section{Literature Review}

\begin{frame}
    \frametitle{Theoretical Foundations}
    \begin{itemize}
        \item \textbf{Credit Migration \& Markov Chains}
        \begin{itemize}
            \item Jarrow, Lando \& Turnbull (1997) --- credit transitions as discrete-time Markov processes.
            \item Bangia et al.\ (2002) --- separate matrices for expansion vs.\ contraction regimes.
        \end{itemize}
        \item \textbf{Regime-Switching Models}
        \begin{itemize}
            \item Hamilton (1989) --- foundational Markov-switching model for business cycle dynamics.
            \item Koopman \& Lucas (2005) --- conditioning migration matrices on macroeconomic indicators improves forecast accuracy by 15--25\%.
            \item Ang \& Timmermann (2012) --- exogenous regime indicators yield more stable and interpretable estimates than latent specifications.
        \end{itemize}
        \item \textbf{IFRS 9 ECL \& Climate Risk}
        \begin{itemize}
            \item IASB (2014) --- forward-looking ECL, incorporation of future economic conditions.
            \item EBA (2023) --- explicit mandate to integrate physical climate risk into ECL models.
            \item Castellani \& Cincinelli (2015) --- 40\% PAR increase in African MFIs during severe drought years.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Literature Gap \& This Study's Position}
    \begin{itemize}
        \item \textbf{What has been done:}
        \begin{itemize}
            \item Endogenous RSMs conditioned on GDP, credit spreads, and financial cycles.
            \item Qualitative climate risk assessments for Sub-Saharan African MFIs.
        \end{itemize}
        \item \textbf{What is missing (four gaps):}
        \begin{itemize}
            \item No \textit{exogenous, climate-driven} RSM applied to credit migration in an African microfinance context.
            \item No SPEI-conditioned two-regime ECL model for Zimbabwe or comparable economies.
            \item No IFRS 9 compliant climate scenario provisioning framework for the MFI sector.
            \item No fully specified structural model for Zimbabwean MFI credit risk.
        \end{itemize}
        \item \textbf{This research fills all four gaps:} Using SPEI-3 as the exogenous regime driver, with direct IFRS 9 ECL output, calibrated to Zimbabwean MFI data.
    \end{itemize}
\end{frame}

\section{Chapter 3: Research Methodology}

\begin{frame}
    \frametitle{Research Design \& Appropriateness}
    \begin{block}{Design: Quantitative Exogenous Regime-Switching Model}
        Justified by the research problem and the literature.
    \end{block}
    \vspace{0.3em}
    \begin{itemize}
        \item \textbf{Why quantitative?} IFRS 9 mandates mathematically rigorous, forward-looking ECL estimates (IASB, 2014); qualitative assessments are non-compliant.
        \item \textbf{Why regime-switching?} Hamilton (1989) and Koopman \& Lucas (2005) prove RSMs outperform static models for credit dynamics with persistent, non-linear shifts --- exactly the behaviour observed during drought events.
        \item \textbf{Why exogenous (SPEI), not endogenous?} Ang \& Timmermann (2012): observable external indicators yield more stable, interpretable estimates with reduced parameter uncertainty compared to latent Markov regimes.
        \item \textbf{Causal validity:} Drought $\to$ defaults (never the reverse) --- satisfies the exogeneity assumption, enabling valid causal inference without simultaneity bias.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Conceptual Framework}
    \begin{center}
    \renewcommand{\arraystretch}{1.4}
    \begin{tabular}{ccc}
        \textbf{Exogenous Driver} & $\longrightarrow$ & \textbf{Regime Classification} \\
        SPEI-3 Drought Index ($\gamma_t$) & & Normal vs.\ Drought-Stress \\[0.2cm]
        & $\downarrow$ & \\[0.2cm]
        \textbf{Regime Weight} & & $\omega(\gamma_t) = \dfrac{1}{1+e^{-\hat{\kappa}(\gamma_t-\hat{\gamma}_0)}}$ \\[0.4cm]
        & $\downarrow$ & \\[0.2cm]
        \textbf{Migration Dynamics} & & $P(\gamma_t) = \omega\,\hat{M}_{\text{stress}} + (1-\omega)\,\hat{M}_{\text{normal}}$ \\[0.2cm]
        & $\downarrow$ & \\[0.2cm]
        \textbf{IFRS 9 ECL Output} & & $\text{ECL} = \text{EAD} \times \text{LGD} \times \text{PD}(\gamma_t)$
    \end{tabular}
    \end{center}
    \vspace{0.3em}
    \begin{itemize}
        \item \textbf{$\omega = 0$}: Fully normal regime \quad $|$ \quad \textbf{$\omega = 1$}: Fully drought-stress regime
        \item Estimated threshold $\hat{\gamma}_0 = -1.10$ corresponds to the WMO moderate drought definition.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Data Sources \& Reliability}
    \begin{itemize}
        \item \textbf{Primary --- Credit Transition Data:}
        \begin{itemize}
            \item Source: Regulatory returns filed by licensed MFIs with the \textbf{Reserve Bank of Zimbabwe (RBZ)}.
            \item Coverage: 6 borrowers, 4 provinces (Mashonaland Central, Manicaland, Matabeleland South, Midlands), 2014Q1--2024Q4.
            \item 59 observations $\Rightarrow$ 51 valid one-quarter transitions after cleaning.
            \item \textit{Reliability:} Official supervisory data; mandatory quarterly reporting ensures completeness.
        \end{itemize}
        \item \textbf{Secondary --- Climate Data:}
        \begin{itemize}
            \item Source: \textbf{Global SPEI Database} (Vicente-Serrano et al., 2010), cross-referenced with Zimbabwe Meteorological Services Department (MSD).
            \item District-level quarterly SPEI-3 values spatially aligned to credit data.
            \item \textit{Reliability:} International standard drought index; peer-reviewed, publicly archived.
        \end{itemize}
        \item \textbf{Data cleaning:} Quarter-end alignment, single-gap interpolation, constant-2024-USD deflation, exclusion of missing fields.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Model Outline \& Estimation}
    \begin{itemize}
        \item \textbf{5-state rating system:} S5 (best) $\to$ S4 $\to$ S3 $\to$ S2 $\to$ S1 (default, absorbing).
        \item \textbf{Conditional transition probability:}
        \[P_{ij}(\gamma_t) = (1-\omega(\gamma_t))\,m_{ij}^{\text{normal}} + \omega(\gamma_t)\,m_{ij}^{\text{stress}}\]
        \item \textbf{MLE log-likelihood:}
        \[\mathcal{L}(\Theta) = \sum_{n,t} \log\!\left[\omega(\gamma_{n,t})\,m_{i,j}^{\text{stress}} + (1-\omega(\gamma_{n,t}))\,m_{i,j}^{\text{normal}}\right]\]
        \item \textbf{Softmax reparameterization:} enforces row-sum $= 1$ and non-negativity.
        \item \textbf{Dirichlet MAP priors} ($\alpha_{\text{diag}}=1.0$, $\alpha_{\text{off}}=0.5$) stabilize sparse regimes.
        \item \textbf{L-BFGS-B optimizer} with 34 free parameters ($K=5$).
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Reliability \& Validity Tests}
    \begin{itemize}
        \item \textbf{Bootstrap Inference ($B = 1{,}000$ replications):}
        \begin{itemize}
            \item Clustered bootstrap (resample borrowers) preserves panel structure.
            \item Yields 95\% CIs for all parameters: $\hat{\kappa}$, $\hat{\gamma}_0$, SVDM.
        \end{itemize}
        \item \textbf{Likelihood Ratio Test (LRT):}
        \[LR = 2(\ell_{\text{two}} - \ell_{\text{single}}) \sim \chi^2(\text{df}), \quad \text{df} = K(K-1)+2 = 22\]
        Formally tests $H_0: \hat{M}_{\text{normal}} = \hat{M}_{\text{stress}}$.
        \item \textbf{SVDM --- Structural Divergence:}
        \[\text{SVDM} = \sum_{k=1}^K |\sigma_k(\hat{M}_{\text{normal}}) - \sigma_k(\hat{M}_{\text{stress}})|\]
        \item \textbf{Out-of-Sample Backtesting (expanding window, last 20\%):}
        \begin{itemize}
            \item AUC (ROC Curve): discrimination; industry threshold $\geq 0.70$.
            \item Brier Score: calibration accuracy; lower is better.
        \end{itemize}
        \item \textbf{Robustness checks:} LGD sensitivity (40\%, 45\%, 50\%); lagged SPEI ($\gamma_{t-1}$).
    \end{itemize}
\end{frame}

\section{Chapter 5: Conclusions \& Recommendations}

\begin{frame}
    \frametitle{Conclusions --- Linked to Objectives \& Literature}
    \begin{enumerate}
        \item \textbf{Time-homogeneous models are structurally inadequate.}\\
              SVDM = 0.309; static model under-provisions by 36--67\%. Confirms Bangia et al.\ (2002): regime-separation materially improves credit risk estimates.
        \item \textbf{SPEI-3 is a valid, interpretable regime indicator.}\\
              $\hat{\gamma}_0 = -1.10$ matches WMO moderate drought. Corroborates Vicente-Serrano et al.\ (2010) on SPEI's superiority for agricultural drought monitoring.
        \item \textbf{The two-regime model improves out-of-sample prediction.}\\
              AUC = 0.724 exceeds the 0.70 threshold; Brier Score improves by 31.7\%. Consistent with Koopman \& Lucas (2005): macro conditioning improves discrimination by 15--25\%.
        \item \textbf{The ECL gap has direct capital adequacy consequences.}\\
              Framework delivers multi-scenario, probability-weighted ECL directly mandated by IFRS 9 (IASB, 2014) and EBA (2023) climate risk guidance, extending prior literature to an African MFI context.
    \end{enumerate}
\end{frame}

\begin{frame}
    \frametitle{Recommendations}
    \begin{itemize}
        \item \textbf{For MFIs:}
        \begin{itemize}
            \item Adopt the two-regime ECL framework as the standard IFRS 9 provisioning model.
            \item Maintain drought-contingency capital buffer $\approx$ 50\% of (severe drought ECL $-$ baseline ECL).
            \item Integrate quarterly SPEI readings into credit review cycles as a regime trigger.
        \end{itemize}
        \item \textbf{For the Reserve Bank of Zimbabwe:}
        \begin{itemize}
            \item Mandate SPEI-conditioned stress testing in prudential guidelines.
            \item Establish a centralised quarterly loan-level credit transition database.
            \item Publish sector-level SPEI--PAR correlations in the Financial Stability Report.
        \end{itemize}
        \item \textbf{For Future Research:}
        \begin{itemize}
            \item Replicate with full RBZ panel ($\approx$50,000 transitions) for definitive LRT resolution.
            \item Extend to three regimes (normal, moderate drought, severe drought).
            \item Benchmark against ML classifiers (Random Forest, Gradient Boosting).
            \item Validate cross-country (Zambia, Malawi, Mozambique) to establish regional generalisability.
        \end{itemize}
    \end{itemize}
\end{frame}
